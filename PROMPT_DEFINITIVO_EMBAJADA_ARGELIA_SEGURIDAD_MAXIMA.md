### 2. Middleware de Seguridad

```typescript
// middleware.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { Ratelimit } from '@upstash/ratelimit';
import { Redis } from '@upstash/redis';

// Rate limiting con Redis
const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(100, '1 m'), // 100 requests por minuto
  analytics: true,
});

// IPs bloqueadas (actualizar dinámicamente)
const blockedIPs = new Set<string>();

// Países permitidos (ISO 3166-1 alpha-2)
const allowedCountries = new Set([
  'ES', 'DZ', 'FR', 'DE', 'IT', 'PT', 'BE', 'NL', 'GB', 
  'US', 'CA', 'MA', 'TN', 'LY', 'EG', 'SA', 'AE', 'QA'
]);

export async function middleware(request: NextRequest) {
  const ip = request.ip ?? request.headers.get('x-forwarded-for') ?? 'unknown';
  const country = request.geo?.country ?? 'unknown';
  const pathname = request.nextUrl.pathname;
  
  // 1. Bloqueo de IPs maliciosas
  if (blockedIPs.has(ip)) {
    return new NextResponse('Access Denied', { status: 403 });
  }
  
  // 2. Geo-blocking (opcional, descomentar si necesario)
  // if (!allowedCountries.has(country)) {
  //   return new NextResponse('Access Denied from your region', { status: 403 });
  // }
  
  // 3. Rate limiting
  const { success, limit, reset, remaining } = await ratelimit.limit(ip);
  
  if (!success) {
    return new NextResponse('Too Many Requests', {
      status: 429,
      headers: {
        'X-RateLimit-Limit': limit.toString(),
        'X-RateLimit-Remaining': remaining.toString(),
        'X-RateLimit-Reset': reset.toString(),
        'Retry-After': Math.ceil((reset - Date.now()) / 1000).toString(),
      },
    });
  }
  
  // 4. Protección de rutas admin
  if (pathname.startsWith('/admin')) {
    const token = request.cookies.get('admin-token');
    if (!token) {
      return NextResponse.redirect(new URL('/login', request.url));
    }
    // Verificar token JWT aquí
  }
  
  // 5. Protección contra path traversal
  if (pathname.includes('..') || pathname.includes('%2e%2e')) {
    return new NextResponse('Invalid Path', { status: 400 });
  }
  
  // 6. Logging de seguridad
  console.log(JSON.stringify({
    timestamp: new Date().toISOString(),
    ip,
    country,
    method: request.method,
    path: pathname,
    userAgent: request.headers.get('user-agent'),
  }));
  
  const response = NextResponse.next();
  
  // Headers adicionales de seguridad
  response.headers.set('X-Request-ID', crypto.randomUUID());
  response.headers.set('X-RateLimit-Remaining', remaining.toString());
  
  return response;
}

export const config = {
  matcher: [
    '/((?!_next/static|_next/image|favicon.ico|public).*)',
  ],
};
```

### 4. Protección CSRF

```typescript
// lib/csrf.ts
import { randomBytes, createHmac } from 'crypto';
import { cookies } from 'next/headers';

const CSRF_SECRET = process.env.CSRF_SECRET!;
const CSRF_COOKIE_NAME = '__csrf';
const CSRF_HEADER_NAME = 'x-csrf-token';

export function generateCSRFToken(): string {
  const token = randomBytes(32).toString('hex');
  const timestamp = Date.now().toString();
  const signature = createHmac('sha256', CSRF_SECRET)
    .update(`${token}:${timestamp}`)
    .digest('hex');
  
  return `${token}:${timestamp}:${signature}`;
}

export function validateCSRFToken(token: string): boolean {
  try {
    const [tokenPart, timestamp, signature] = token.split(':');
    
    // Verificar que el token no ha expirado (1 hora)
    const tokenAge = Date.now() - parseInt(timestamp);
    if (tokenAge > 3600000) return false;
    
    // Verificar firma
    const expectedSignature = createHmac('sha256', CSRF_SECRET)
      .update(`${tokenPart}:${timestamp}`)
      .digest('hex');
    
    return signature === expectedSignature;
  } catch {
    return false;
  }
}

export function setCSRFCookie(token: string): void {
  cookies().set(CSRF_COOKIE_NAME, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === 'production',
    sameSite: 'strict',
    maxAge: 3600, // 1 hora
    path: '/',
  });
}
```

```typescript
// app/api/contact/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { contactFormSchema } from '@/lib/validations/contact';
import { validateCSRFToken } from '@/lib/csrf';
import { verifyHCaptcha } from '@/lib/hcaptcha';
import { ratelimit } from '@/lib/ratelimit';
import { logSecurityEvent } from '@/lib/audit';

export async function POST(request: NextRequest) {
  const ip = request.ip ?? 'unknown';
  const startTime = Date.now();
  
  try {
    // 1. Rate limiting específico para este endpoint
    const { success } = await ratelimit.limit(`contact:${ip}`);
    if (!success) {
      await logSecurityEvent('RATE_LIMIT_EXCEEDED', { ip, endpoint: '/api/contact' });
      return NextResponse.json(
        { error: 'Demasiadas solicitudes. Intente más tarde.' },
        { status: 429 }
      );
    }
    
    // 2. Verificar CSRF
    const csrfToken = request.headers.get('x-csrf-token');
    if (!csrfToken || !validateCSRFToken(csrfToken)) {
      await logSecurityEvent('CSRF_VALIDATION_FAILED', { ip });
      return NextResponse.json(
        { error: 'Token de seguridad inválido' },
        { status: 403 }
      );
    }
    
    // 3. Parsear y validar body
    const body = await request.json();
    const validationResult = contactFormSchema.safeParse(body);
    
    if (!validationResult.success) {
      await logSecurityEvent('VALIDATION_FAILED', { 
        ip, 
        errors: validationResult.error.errors 
      });
      return NextResponse.json(
        { error: 'Datos inválidos', details: validationResult.error.errors },
        { status: 400 }
      );
    }
    
    const data = validationResult.data;
    
    // 4. Verificar honeypot
    if (data.website && data.website.length > 0) {
      await logSecurityEvent('HONEYPOT_TRIGGERED', { ip });
      // Simular éxito para confundir bots
      return NextResponse.json({ success: true });
    }
    
    // 5. Verificar hCaptcha
    const captchaValid = await verifyHCaptcha(data.captchaToken);
    if (!captchaValid) {
      await logSecurityEvent('CAPTCHA_FAILED', { ip });
      return NextResponse.json(
        { error: 'Verificación de captcha fallida' },
        { status: 400 }
      );
    }
    
    // 6. Procesar el formulario (guardar en DB, enviar email, etc.)
    // ... lógica de negocio ...
    
    // 7. Log de éxito
    await logSecurityEvent('CONTACT_FORM_SUCCESS', {
      ip,
      asunto: data.asunto,
      processingTime: Date.now() - startTime,
    });
    
    return NextResponse.json({ 
      success: true,
      message: 'Mensaje enviado correctamente'
    });
    
  } catch (error) {
    await logSecurityEvent('CONTACT_FORM_ERROR', { 
      ip, 
      error: error instanceof Error ? error.message : 'Unknown error'
    });
    
    return NextResponse.json(
      { error: 'Error interno del servidor' },
      { status: 500 }
    );
  }
}
```

### 6. Audit Logging

```typescript
// lib/audit.ts
import { Redis } from '@upstash/redis';

const redis = Redis.fromEnv();

export type SecurityEventType =
  | 'LOGIN_SUCCESS'
  | 'LOGIN_FAILED'
  | 'LOGOUT'
  | 'RATE_LIMIT_EXCEEDED'
  | 'CSRF_VALIDATION_FAILED'
  | 'VALIDATION_FAILED'
  | 'HONEYPOT_TRIGGERED'
  | 'CAPTCHA_FAILED'
  | 'CONTACT_FORM_SUCCESS'
  | 'CONTACT_FORM_ERROR'
  | 'ADMIN_ACCESS'
  | 'CONTENT_MODIFIED'
  | 'SUSPICIOUS_ACTIVITY';

interface SecurityEvent {
  type: SecurityEventType;
  timestamp: string;
  ip?: string;
  userId?: string;
  userAgent?: string;
  metadata?: Record<string, unknown>;
}

export async function logSecurityEvent(
  type: SecurityEventType,
  metadata?: Record<string, unknown>
): Promise<void> {
  const event: SecurityEvent = {
    type,
    timestamp: new Date().toISOString(),
    ...metadata,
  };
  
  // Guardar en Redis con TTL de 90 días
  const key = `audit:${Date.now()}:${crypto.randomUUID()}`;
  await redis.set(key, JSON.stringify(event), { ex: 7776000 }); // 90 días
  
  // También enviar a sistema SIEM si está configurado
  if (process.env.SIEM_WEBHOOK_URL) {
    fetch(process.env.SIEM_WEBHOOK_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(event),
    }).catch(console.error);
  }
  
  // Log para eventos críticos
  if (['LOGIN_FAILED', 'CSRF_VALIDATION_FAILED', 'SUSPICIOUS_ACTIVITY'].includes(type)) {
    console.error('SECURITY ALERT:', JSON.stringify(event));
  }
}
```
